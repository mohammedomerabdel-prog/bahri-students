class FinancialSecretariat: def __init__(self, initial_balance=0.0): self.balance = initial_balance self.transactions = [] def add_income(self, amount, description): """تسجيل إيراد (دعم، اشتراكات، تبرعات)""" self.balance += amount self.transactions.append({"type": "إيراد", "amount": amount, "desc": description}) print(f"تم إضافة إيراد: +{amount} ({description})") def add_expense(self, amount, description): """تسجيل منصرف (مصروفات أنشطة، طباعة، إلخ)""" if amount > self.balance: print("تحذير: الرصيد غير كافٍ لإتمام هذا المنصرف!") return False self.balance -= amount self.transactions.append({"type": "منصرف", "amount": amount, "desc": description}) print(f"تم تسجيل منصرف: -{amount} ({description})") return True def get_statement(self): """عرض كشف الحساب والتقرير المالي""" print("\n--- كشف الحساب المالي للأمانة ---") for t in self.transactions: print(f"[{t['type']}] {t['amount']} - {t['desc']}") print(f"الرصيد الحالي المتبقي: {self.balance}\n") # --- مثال على الاستخدام --- if __name__ == "__main__": # إنشاء خزينة الأمانة المالية برصيد ابتدائي treasury = FinancialSecretariat(initial_balance=50000) # تسجيل بعض المعاملات treasury.add_income(25000, "اشتراكات الأعضاء لشهر أكتوبر") treasury.add_expense(15000, "تكلفة طباعة الملخصات والمستندات") # عرض التقرير المالي النهائي treasury.get_statement()
